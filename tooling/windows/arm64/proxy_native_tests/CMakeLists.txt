cmake_minimum_required(VERSION 3.14)
project(fengwo_proxy_native_verification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PROXY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../plugins/proxy/windows")
add_executable(proxy_session_test "${PROXY_SOURCE}/test/proxy_session_test.cpp")
target_compile_definitions(proxy_session_test PRIVATE FENGWO_PROXY_SESSION_STANDALONE)
add_executable(proxy_session_windows_test "${PROXY_SOURCE}/test/proxy_session_windows_test.cpp")
target_link_libraries(proxy_session_windows_test PRIVATE wininet advapi32)
add_executable(proxy_winhttp_compatibility_test "${PROXY_SOURCE}/test/proxy_winhttp_compatibility_test.cpp")
target_link_libraries(proxy_winhttp_compatibility_test PRIVATE wininet winhttp)
foreach(TARGET proxy_session_test proxy_session_windows_test proxy_winhttp_compatibility_test)
  target_compile_options(${TARGET} PRIVATE /W4 /WX /EHsc /utf-8)
endforeach()
