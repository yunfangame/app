cmake_minimum_required(VERSION 3.14)
project(fengwo_proxy_snapshot_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(proxy_session_test ../proxy_session_test.cpp)
target_compile_definitions(proxy_session_test PRIVATE FENGWO_PROXY_SESSION_STANDALONE)
add_executable(proxy_session_windows_test ../proxy_session_windows_test.cpp)
target_link_libraries(proxy_session_windows_test PRIVATE wininet advapi32)
if(MSVC)
  target_compile_options(proxy_session_test PRIVATE /W4 /WX)
  target_compile_options(proxy_session_windows_test PRIVATE /W4 /WX)
endif()
add_test(NAME session COMMAND proxy_session_test)
add_test(NAME actual_wininet_snapshot COMMAND proxy_session_windows_test)
set_tests_properties(actual_wininet_snapshot PROPERTIES RUN_SERIAL TRUE)
